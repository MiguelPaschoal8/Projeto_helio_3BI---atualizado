DROP SCHEMA IF EXISTS Restaurante;

CREATE SCHEMA IF NOT EXISTS Restaurante DEFAULT CHARACTER SET utf8;
USE Restaurante;

DROP TABLE IF EXISTS Pagamento;
DROP TABLE IF EXISTS ItemPedido;
DROP TABLE IF EXISTS Pedido;
DROP TABLE IF EXISTS Produto;   
DROP TABLE IF EXISTS Categoria;
DROP TABLE IF EXISTS Mesa;
DROP TABLE IF EXISTS Funcionario;
DROP TABLE IF EXISTS Cliente;

CREATE TABLE IF NOT EXISTS Cliente(
    idCliente INT PRIMARY KEY AUTO_INCREMENT,
    nomeCliente VARCHAR(100) NOT NULL,
    telefone VARCHAR(20) UNIQUE,
    email VARCHAR(100) UNIQUE,
    cpf CHAR(11) UNIQUE NOT NULL
) ENGINE = InnoDB;

CREATE TABLE IF NOT EXISTS Funcionario(
    idFuncionario INT PRIMARY KEY AUTO_INCREMENT,
    nomeFuncionario VARCHAR(100) NOT NULL,
    Cargo ENUM('Garçom','Caixa','Cozinheiro','Gerente') NOT NULL,
    salario DECIMAL(10,2) NOT NULL,
    senha VARCHAR(255) NOT NULL,
    CHECK(salario > 0)
) ENGINE = InnoDB;

CREATE TABLE IF NOT EXISTS Mesa(
    idMesa INT PRIMARY KEY AUTO_INCREMENT,
    numeroMesa INT UNIQUE NOT NULL,
    capacidade INT NOT NULL,
    statusMesa ENUM('Livre','Ocupada','Reservada') DEFAULT 'Livre',
    CHECK(capacidade > 0)
) ENGINE = InnoDB;

CREATE TABLE IF NOT EXISTS Categoria(
    idCategoria INT PRIMARY KEY AUTO_INCREMENT,
    nomeCategoria VARCHAR(50) UNIQUE NOT NULL
) ENGINE = InnoDB;

CREATE TABLE IF NOT EXISTS Produto(
    idProduto INT PRIMARY KEY AUTO_INCREMENT,
    nomeProduto VARCHAR(100) UNIQUE NOT NULL,
    preco DECIMAL(10,2) NOT NULL,
    estoque INT UNSIGNED DEFAULT 0,
    idCategoria INT,
    CHECK(preco > 0),
    INDEX idxCategoria(idCategoria ASC),
    FOREIGN KEY(idCategoria) REFERENCES Categoria(idCategoria)
) ENGINE = InnoDB;

CREATE TABLE IF NOT EXISTS Pedido(
    idPedido INT PRIMARY KEY AUTO_INCREMENT,
    dataPedido DATETIME DEFAULT CURRENT_TIMESTAMP,
    statusPedido ENUM('Em preparo','Entregue','Cancelado') DEFAULT 'Em preparo',
    idCliente INT NULL,  -- Alterado para NULL por causa do SET NULL
    idFuncionario INT NULL,  -- Alterado para NULL
    idMesa INT NULL,  -- Alterado para NULL
    INDEX idxCliente(idCliente ASC),
    INDEX idxFuncionario(idFuncionario ASC),
    INDEX idxMesa(idMesa ASC),
    INDEX idxDataPedido(dataPedido ASC),
    FOREIGN KEY(idCliente) REFERENCES Cliente(idCliente) ON DELETE SET NULL ON UPDATE CASCADE,
    FOREIGN KEY(idFuncionario) REFERENCES Funcionario(idFuncionario) ON DELETE SET NULL ON UPDATE CASCADE,
    FOREIGN KEY(idMesa) REFERENCES Mesa(idMesa) ON DELETE SET NULL ON UPDATE CASCADE
) ENGINE = InnoDB;

CREATE TABLE IF NOT EXISTS ItemPedido(
    idItem INT PRIMARY KEY AUTO_INCREMENT,
    idPedido INT NOT NULL,
    idProduto INT NOT NULL,
    quantidade INT NOT NULL,
    subtotal DECIMAL(10,2) NOT NULL,
    CHECK(quantidade > 0),
    CHECK(subtotal > 0),
    INDEX idxPedido(idPedido ASC),
    INDEX idxProduto(idProduto ASC),
    FOREIGN KEY(idPedido) REFERENCES Pedido(idPedido),
    FOREIGN KEY(idProduto) REFERENCES Produto(idProduto)
) ENGINE = InnoDB;

CREATE TABLE IF NOT EXISTS Pagamento(
    idPagamento INT PRIMARY KEY AUTO_INCREMENT,
    idPedido INT UNIQUE NOT NULL,
    formaPagamento ENUM('Dinheiro','Cartão','PIX','Pendente') DEFAULT 'Pendente',
    valor DECIMAL(10,2) NOT NULL,
    statusPagamento ENUM('Pendente','Pago') DEFAULT 'Pendente',
    dataPagamento DATETIME DEFAULT CURRENT_TIMESTAMP,
    CHECK(valor > 0),
    INDEX idxDataPagamento(dataPagamento ASC),
    FOREIGN KEY(idPedido) REFERENCES Pedido(idPedido)
	ON DELETE CASCADE 
) ENGINE = InnoDB;
